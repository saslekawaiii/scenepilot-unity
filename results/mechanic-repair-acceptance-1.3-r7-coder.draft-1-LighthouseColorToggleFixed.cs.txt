using UnityEngine;

public class LighthouseColorToggleFixed : MonoBehaviour
{
    private Color currentColor = new Color(0.235f, 0.286f, 1f);
    private Color alternateColor = new Color(1f, 0.749f, 0.11f);

    void Start()
    {
        Renderer renderer = GetComponent<Renderer>();
        if (renderer != null)
        {
            renderer.material.color = currentColor;
        }
    }

    void Update()
    {
        if (Input.GetKeyDown(KeyCode.F))
        {
            ToggleColor();
        }
    }

    private void ToggleColor()
    {
        Renderer renderer = GetComponent<Renderer>();
        if (renderer != null)
        {
            Color nextColor = currentColor == alternateColor ? currentColor : alternateColor;
            renderer.material.color = nextColor;
            currentColor = nextColor;
        }
    }
}
