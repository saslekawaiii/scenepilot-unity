using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Runtime.CompilerServices;
using System.Security.Cryptography;
using System.Text;
using Newtonsoft.Json.Linq;
using UnityEditor;
using UnityEngine;
using UnityEngine.UIElements;
using WorldWeaver;
using Object = UnityEngine.Object;

/// <summary>Read-only snapshot of one actual, already open native Play Mode conversation.</summary>
public static class NativeConversationUiAudit
{
    [MenuItem("WorldWeaver/Acceptance/CaptureOpenConversationUi")]
    public static void CaptureOpenConversationUi()
    {
        Require(Application.isPlaying && !Application.isBatchMode, "Use the normal GUI Editor in Play Mode.");
        var players = Object.FindObjectsByType<WorldPlayer>(FindObjectsSortMode.None).Where(p => p.isActiveAndEnabled && p.Conversation && p.Conversation.IsOpen).ToArray();
        Require(players.Length == 1, "Exactly one actual WorldPlayer must already have an open conversation.");
        var player = players[0]; var ui = player.Conversation; var npc = ui.ActiveNpc;
        var document = ui.GetComponent<UIDocument>(); var root = document ? document.rootVisualElement : null;
        Require(npc && root != null && root.panel != null, "The existing conversation needs an active NPC and attached UIDocument.");
        var panel = root.Q(className: "conversation-panel"); var composer = root.Q(className: "reply-composer");
        var input = root.Q<TextField>("conversation-input"); var send = root.Q<Button>("conversation-send"); var close = root.Q<Button>(className: "close-button");
        var header = root.Q(className: "conversation-header"); var history = root.Q("conversation-history"); var profile = root.Q(className: "character-card");
        var surface = input?.Q(className: "unity-text-field__input"); var focused = root.focusController?.focusedElement as VisualElement;
        var viewport = root.worldBound; bool compact = root.ClassListContains("compact");
        var elements = new Dictionary<string, VisualElement> { ["viewport"] = root, ["backdrop"] = root.Q("conversation-backdrop"), ["stage"] = root.Q(className: "dialogue-stage"),
            ["profile"] = profile, ["panel"] = panel, ["header"] = header, ["history"] = history, ["composer"] = composer, ["input"] = input, ["inputSurface"] = surface, ["send"] = send, ["close"] = close };
        var geometry = new JObject(); foreach (var pair in elements) geometry[pair.Key] = Element(pair.Value);
        var checks = new JObject {
            ["viewportHasFinitePositiveBounds"] = Valid(viewport),
            ["coreControlsVisible"] = new VisualElement[] { panel, composer, input, surface, send, close, header, history }.All(Visible),
            ["coreControlsInsideViewport"] = new VisualElement[] { panel, composer, input, surface, send, close, header, history }.All(e => Inside(root, e)),
            ["controlsInsideConversationPanel"] = new VisualElement[] { composer, input, surface, send, close, header, history }.All(e => Inside(panel, e)),
            ["inputAndSendInsideComposer"] = Inside(composer, input) && Inside(composer, surface) && Inside(composer, send),
            ["inputAndSendDoNotOverlap"] = Separate(input, send), ["closeAndInputDoNotOverlap"] = Separate(close, input),
            ["closeAndSendDoNotOverlap"] = Separate(close, send), ["headerAndComposerDoNotOverlap"] = Separate(header, composer),
            ["historyAndComposerDoNotOverlap"] = Separate(history, composer),
            ["compactClassMatchesActualViewport"] = compact == (viewport.width < 900),
            ["profileVisibilityMatchesLayout"] = compact ? !Visible(profile) : Visible(profile),
            ["actualMultilineInputAvailable"] = input != null && input.multiline && input.maxLength == 1600,
            ["documentTargetsGameViewport"] = document.panelSettings && !document.panelSettings.targetTexture
        };
        string value = input?.value ?? "", source = SourcePath(); Require(File.Exists(source), "The installed audit source file must be readable for its SHA256.");
        var report = new JObject { ["kind"] = "actual-open-native-conversation-ui-read-only-audit", ["recordedUtc"] = DateTime.UtcNow.ToString("o"), ["unity"] = Application.unityVersion,
            ["sourceSha256"] = Hash(File.ReadAllBytes(source)), ["isPlaying"] = Application.isPlaying, ["layoutMode"] = compact ? "compact" : "wide", ["coordinateSpace"] = "UI Toolkit worldBound panel points; not desktop screen coordinates",
            ["screenPixels"] = new JObject { ["width"] = Screen.width, ["height"] = Screen.height }, ["panelScale"] = document.panelSettings ? document.panelSettings.scale : 0,
            ["playerInstanceId"] = player.GetInstanceID(), ["npcInstanceId"] = npc.GetInstanceID(), ["npcCharacterIdSha256"] = Hash(Encoding.UTF8.GetBytes(npc.CharacterId ?? "")),
            ["npcNameSha256"] = Hash(Encoding.UTF8.GetBytes(npc.NpcName ?? "")), ["npcAiEnabled"] = npc.AiEnabled, ["npcBusy"] = npc.Busy, ["historyEntryCount"] = npc.ConversationHistory.Count,
            ["input"] = new JObject { ["characterCount"] = value.Length, ["sha256"] = Hash(Encoding.UTF8.GetBytes(value)), ["containsNewline"] = value.Contains("\n"), ["enabled"] = input != null && input.enabledInHierarchy,
                ["hasFocusWithinInput"] = input != null && focused != null && (focused == input || input.Contains(focused)) },
            ["sendEnabled"] = send != null && send.enabledInHierarchy, ["elements"] = geometry, ["checks"] = checks, ["passed"] = checks.Properties().All(p => (bool)p.Value),
            ["openedUiByAudit"] = false, ["requestedInferenceByAudit"] = false, ["sentMessageByAudit"] = false, ["changedUiByAudit"] = false, ["nativeKeyboardInputProven"] = false,
            ["scope"] = "One observed open layout only. Input content is recorded as length/SHA256, without dialogue text or NPC names. Native keyboard and screenshot evidence must be captured separately." };
        string output = OutputDirectory(); Directory.CreateDirectory(output);
        string path = Path.Combine(output, "native-conversation-ui-" + DateTime.UtcNow.ToString("yyyyMMddTHHmmssfffffffZ") + "-" + Guid.NewGuid().ToString("N").Substring(0, 8) + ".json");
        using (var file = new FileStream(path, FileMode.CreateNew, FileAccess.Write)) using (var writer = new StreamWriter(file, new UTF8Encoding(false))) writer.Write(report.ToString());
        Debug.Log("Read-only native conversation UI audit: " + path + "; geometry passed=" + (bool)report["passed"]);
    }
    [MenuItem("WorldWeaver/Acceptance/CaptureOpenConversationUi", true)]
    static bool CanCapture() => Application.isPlaying && !Application.isBatchMode;
    static JObject Element(VisualElement element)
    {
        if (element == null) return new JObject { ["exists"] = false };
        var b = element.worldBound; return new JObject { ["exists"] = true, ["name"] = element.name, ["type"] = element.GetType().Name, ["visible"] = Visible(element), ["enabled"] = element.enabledInHierarchy,
            ["display"] = element.resolvedStyle.display.ToString(), ["visibility"] = element.resolvedStyle.visibility.ToString(), ["bounds"] = new JObject { ["x"] = Number(b.x), ["y"] = Number(b.y), ["width"] = Number(b.width), ["height"] = Number(b.height) } };
    }
    static JToken Number(float number) => float.IsNaN(number) || float.IsInfinity(number) ? JValue.CreateNull() : new JValue(number);
    static bool Valid(Rect rect) => new[] { rect.x, rect.y, rect.width, rect.height }.All(v => !float.IsNaN(v) && !float.IsInfinity(v)) && rect.width > .5f && rect.height > .5f;
    static bool Visible(VisualElement element)
    {
        if (element == null || element.panel == null || !Valid(element.worldBound)) return false;
        for (var p = element; p != null; p = p.parent) if (p.resolvedStyle.display == DisplayStyle.None || p.resolvedStyle.visibility != Visibility.Visible || p.resolvedStyle.opacity <= .001f) return false;
        return true;
    }
    static bool Inside(VisualElement parent, VisualElement child)
    {
        if (parent == null || child == null || !Valid(parent.worldBound) || !Valid(child.worldBound)) return false;
        var p = parent.worldBound; var c = child.worldBound; const float tolerance = .75f;
        return c.xMin >= p.xMin - tolerance && c.yMin >= p.yMin - tolerance && c.xMax <= p.xMax + tolerance && c.yMax <= p.yMax + tolerance;
    }
    static bool Separate(VisualElement first, VisualElement second)
    {
        if (first == null || second == null || !Valid(first.worldBound) || !Valid(second.worldBound)) return false;
        var a = first.worldBound; var b = second.worldBound; return Mathf.Min(a.xMax, b.xMax) - Mathf.Max(a.xMin, b.xMin) <= .75f || Mathf.Min(a.yMax, b.yMax) - Mathf.Max(a.yMin, b.yMin) <= .75f;
    }
    static string OutputDirectory()
    {
        string output = Environment.GetEnvironmentVariable("WW_NATIVE_UI_OUTPUT_DIR");
        if (string.IsNullOrWhiteSpace(output))
        {
            var project = Directory.GetParent(Application.dataPath); var repo = project?.Parent;
            Require(project?.Name == "UnityProject" && repo != null && Directory.Exists(Path.Combine(repo.FullName, "results")), "Set WW_NATIVE_UI_OUTPUT_DIR explicitly outside the project Assets/Packages.");
            output = Path.Combine(repo.FullName, "results/screenshots-1.3");
        }
        output = Path.GetFullPath(output); var projectRoot = Directory.GetParent(Application.dataPath).FullName;
        Require(!Under(output, Application.dataPath) && !Under(output, Path.Combine(projectRoot, "Packages")), "Audit JSON must stay outside project Assets/Packages."); return output;
    }
    static bool Under(string path, string parent) => path == parent || path.StartsWith(parent.TrimEnd(Path.DirectorySeparatorChar) + Path.DirectorySeparatorChar, StringComparison.Ordinal);
    static string SourcePath([CallerFilePath] string path = "") => Path.GetFullPath(path);
    static string Hash(byte[] bytes) { using (var hash = SHA256.Create()) return BitConverter.ToString(hash.ComputeHash(bytes)).Replace("-", "").ToLowerInvariant(); }
    static void Require(bool condition, string message) { if (!condition) throw new InvalidOperationException(message); }
}
